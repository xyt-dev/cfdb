const int maxn = 200500;
const int inf = ( 2e9 ) + 2;

vector < pair < pair < int, int >, pair < int, int > > > queries;
int bestCost[maxn];

int l[maxn];
int r[maxn];
int cost[maxn];

int solution( int n, int needLen ) {
    queries.clear();
    for ( int j = 0; j < n; j++ ) {
        queries.pb( mp( mp( l[j], -1 ), mp( r[j], cost[j] ) ) );
        queries.pb( mp( mp( r[j], 1 ), mp( l[j], cost[j] ) ) );
    }
    for ( int j = 0; j < maxn; j++ )
        bestCost[j] = inf;
    ll ans = 2LL * inf;
    sort( queries.begin(), queries.end() );
    int sz = queries.size();
    for ( int j = 0; j < sz; j++ ) {
        int type = queries[j].f.s;
        if ( type == -1 ) {
            int curLen = queries[j].s.f - queries[j].f.f + 1;
            if ( curLen <= needLen )
                ans = min( ans, 1LL * queries[j].s.s + 1LL * bestCost[needLen - curLen] );
        }
        if ( type == 1 ) {
            int curLen = queries[j].f.f - queries[j].s.f + 1;
            bestCost[curLen] = min( bestCost[curLen], queries[j].s.s );
        }
    }
    return ans >= inf ? -1 : ans;
}

## 822E - Liar

You need to determine whether condition *f*(*s*, *t*) ≤ *x* is satisfied. *f*(*s*, *t*) is equal to the minimal number of non-intersect substrings of string *s* which can be concatenated together to get *t*. Substrings should be concatenated in the same order they appear in *s*.

Note that for short strings we can use *DP*(*prefS*, *prefT*) = *f*(*s*[1... *prefS*], *t*[1... *prefT*]).

             classes = new int[maxH][];
             for (int h = 0; h < maxH; h++) {
                 classes[h] = new int[s.length()];

                 int valuesLen = 0;

                     int leftPart = classes[h - 1][i];
                     int rightPart = classes[h - 1][i + (1 << (h - 1))];
                     long curValue = ((long)leftPart << 30) ^ rightPart;
                     values[valuesLen++] = curValue;

                 values = unique(values);

                     int leftPart = classes[h - 1][i];
                     int rightPart = classes[h - 1][i + (1 << (h - 1))];
                     long curValue = ((long)leftPart << 30) ^ rightPart;
                     classes[h][i] = Arrays.binarySearch(values, curValue);
                 }
